# ==============================================================================
# Colemailer 3.4.10 — пример файла config.txt
# ==============================================================================
# Скопируй этот файл в config.txt рядом с bot.py и впиши свои данные:
#   Windows:    copy config.example.txt config.txt
#   Linux/mac:  cp config.example.txt config.txt
#
# Формат: `ключ = значение`, по одной паре на строку. Секции вида [account1]
# нужны только для нескольких аккаунтов. Строки с # игнорируются, кавычки
# вокруг значения парсер срезает, незнакомые ключи молча пропускает.
#
# Проверить, что бот видит конфиг, ничего не запуская:
#   python bot.py --check
#
# ВАЖНО: либо один аккаунт без секций, либо только секции. Если в файле есть
# хотя бы одна секция [account...], ключи вне секций (кроме log_errors_only)
# не читаются вообще.
# ==============================================================================


# ------------------------------------------------------------------ логирование
# true  — в консоль идут только баннер старта и ошибки (флудвейт, обрывы связи,
#         запрет писать в чат). Самый спокойный режим, для 24/7 берут его.
# false — подробный лог: каждый подсчитанный пинг, распознанные команды,
#         решения антиспама, синхронизация папок. Полезно, когда что-то не работает.
log_errors_only = true


# ==============================================================================
# ВАРИАНТ 1 — ОДИН АККАУНТ (ключи пишутся прямо в файле, без секции)
# ==============================================================================

# api_id и api_hash — ключи приложения Telegram. Это не токен бота и не пароль.
# Где взять: https://my.telegram.org -> API development tools (пошагово — в README,
# раздел «Где взять api_id и api_hash»).
# Автоматически, без сайта: python bot.py --auto-api +79991234567
# api_id — только цифры, api_hash — 32 символа латиницей.
api_id = 12345678
api_hash = 0123456789abcdef0123456789abcdef

# Номер аккаунта в международном формате, с плюсом и кодом страны.
# Нужен только для первого входа: потом бот входит по файлу сессии.
phone = +79991234567

# Имя файла сессии (без расширения), лежит рядом с bot.py.
# Сессия — это уже выполненный вход: кто получил файл, тот получил доступ
# к аккаунту. Бэкапить — только вместе с пониманием этого.
session_name = session_userbot

# Папка Telegram, куда складываются каналы, в которые бот вступил по требованию
# админ-ботов с капчей. Папки нет — бот создаст её сам.
auto_sub_folder = автосабнутое

# Гасить спам-пинги и явную рекламу: снять упоминание, пометить прочитанным,
# убрать уведомление. Личные чаты не трогаются никогда.
# false — бот не лезет во входящие вообще (рассылки и команды работают).
auto_read_spam_pings = true

# Авто-подписка на каналы, которых требуют капча-боты: вступить, заглушить,
# положить в auto_sub_folder, нажать «проверить». Работает только для сообщений,
# где бота реально упомянули.
# false — бот ничего не подписывает и чужие кнопки не нажимает.
auto_sub_enabled = true

# Порядок и регистр ключей не важны, пробелы вокруг = допустимы.
# Мьюты (`.мутить`), кнопки (`.кнопки`), превью (`.превью`), рассылки и папки —
# это команды в Telegram, а не настройки конфига. Смотри README, раздел «Команды».


# ==============================================================================
# ВАРИАНТ 2 — НЕСКОЛЬКО АККАУНТОВ (каждый в своей секции)
# ==============================================================================
# Как перейти на этот вариант:
#   1. Убери (или закомментируй) api_* / phone / session_name / auto_* из
#      варианта 1 — иначе получится мешанина, которая не читается.
#   2. Раскомментируй блоки [account1] и [account2], при желании добавь [account3].
#   3. log_errors_only оставь вне секций: он общий на весь файл.
#
# Аккаунты работают параллельно в одном процессе: у каждого своя сессия, свой
# планировщик и своя привязка рассылок. База шаблонов общая — пост, собранный
# на одном аккаунте, виден и второму.
# Второй экземпляр на тот же аккаунт бот не запустит (файловый лок) — так
# задумано, иначе один и тот же пост уходит дважды.
# ==============================================================================
